==== 4.6.1 API Interceptor (Base URL + Auth)

`src/app/core/interceptors/api.interceptor.ts`:

[source,typescript]
----
import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { environment } from '../../../environments/environment';

export const apiInterceptor: HttpInterceptorFn = (req, next) => {
  // Nur Requests zu relativen URLs modifizieren
  if (!req.url.startsWith('http')) {
    const apiReq = req.clone({
      url: `${environment.apiBaseUrl}${req.url}`
    });
    return next(apiReq);
  }
  return next(req);
};
----

==== 4.6.2 Error Interceptor (Globales Error Handling)

`src/app/core/interceptors/error.interceptor.ts`:

[source,typescript]
----
import { HttpInterceptorFn, HttpErrorResponse } from '@angular/common/http';
import { inject } from '@angular/core';
import { catchError, throwError } from 'rxjs';
import { NotificationService } from '../services/notification.service';

export const errorInterceptor: HttpInterceptorFn = (req, next) => {
  const notificationService = inject(NotificationService);

  return next(req).pipe(
    catchError((error: HttpErrorResponse) => {
      let errorMessage = 'Ein unbekannter Fehler ist aufgetreten';

      if (error.error instanceof ErrorEvent) {
        // Client-seitiger Fehler
        errorMessage = `Fehler: ${error.error.message}`;
      } else {
        // Server-seitiger Fehler
        errorMessage = error.error?.message || `Server Fehler: ${error.status}`;
      }

      notificationService.showError(errorMessage);
      return throwError(() => error);
    })
  );
};
----

==== 4.6.3 Loading Interceptor (Loading State)

`src/app/core/interceptors/loading.interceptor.ts`:

[source,typescript]
----
import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { finalize } from 'rxjs';
import { LoadingService } from '../services/loading.service';

export const loadingInterceptor: HttpInterceptorFn = (req, next) => {
  const loadingService = inject(LoadingService);

  loadingService.show();

  return next(req).pipe(
    finalize(() => loadingService.hide())
  );
};
----

==== 4.7.2 Loading Service (Globaler Loading State)

`src/app/core/services/loading.service.ts`:

[source,typescript]
----
import { Injectable, signal } from '@angular/core';

@Injectable({
  providedIn: 'root'
})
export class LoadingService {
  private loadingCount = 0;
  public isLoading = signal<boolean>(false);

  show() {
    this.loadingCount++;
    this.isLoading.set(true);
  }

  hide() {
    this.loadingCount = Math.max(0, this.loadingCount - 1);
    if (this.loadingCount === 0) {
      this.isLoading.set(false);
    }
  }
}
----

**Automatisierungsgrad:** 40% (LLM kann Template generieren)

[source,typescript]

import { Injectable, inject } from '@angular/core';
import { Observable } from 'rxjs';
import { NotesService, NoteDTO } from '../../../api';

@Injectable({
  providedIn: 'root'
})
export class NoteService {
  private apiService = inject(NotesService);

  getAllNotes(): Observable<NoteDTO[]> {
    return this.apiService.listAll();
  }

  getNoteById(id: number): Observable<NoteDTO> {
    return this.apiService.getById(id);
  }

  createNote(note: NoteDTO): Observable<NoteDTO> {
    return this.apiService.create(note);
  }

  updateNote(id: number, note: NoteDTO): Observable<NoteDTO> {
    return this.apiService.update(id, note);
  }

  deleteNote(id: number): Observable<void> {
    return this.apiService.delete(id);
  }

  getNoteHistory(id: number): Observable<any[]> {
    return this.apiService.getHistory(id);
  }
}

== Troubleshooting

=== Häufige Probleme

**1. CORS-Fehler**

[source]
----
Lösung: Prüfe application.properties in Quarkus
Stelle sicher: quarkus.http.cors.origins=http://localhost:4200
----

**2. OpenAPI Client Import-Fehler**

[source]
----
Lösung: Prüfe, ob BASE_PATH in Configuration richtig gesetzt ist
Environment-Variable verwenden: environment.apiBaseUrl
----

**3. Material Styles nicht geladen**

[source]
----
Lösung: src/styles.scss prüfen
@import '@angular/material/prebuilt-themes/indigo-pink.css';
Material Icons und Roboto Font müssen geladen sein
----

**4. Interceptors werden nicht ausgeführt**

[source]
----
Lösung: app.config.ts prüfen
withInterceptors([...]) muss in provideHttpClient() sein
----
