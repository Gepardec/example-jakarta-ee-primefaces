== Zusammenfassung

Dieses Dokument zeigt die Migrationsstrategie für das Notizblock-Projekt von einer Jakarta EE 10 + JSF/Primefaces Anwendung zu einer modernen Architektur mit Quarkus (REST API Backend) und Angular (SPA Frontend).

**Zielsetzung:** Maximale Automatisierung bei minimaler LLM-Nutzung (nur für Angular-Komponenten-Migration).

**Hauptergebnisse:**

* ✅ *Backend-Migration (Jakarta EE zu Quarkus):* Hoher Automatisierungsgrad möglich (~70–80 %)
* ⚠️  *Frontend-Migration (JSF/Primefaces zu Angular):* Mittlerer Automatisierungsgrad (~40–50 % mit LLM)

*Beispiel REST Resource (NoteResource.java):*

[source,java]
----
package com.gepardec.notizblock.resource;

import com.gepardec.notizblock.entity.Note;
import jakarta.transaction.Transactional;
import jakarta.ws.rs.*;
import jakarta.ws.rs.core.MediaType;
import jakarta.ws.rs.core.Response;
import java.util.List;

@Path("/api/notes")
@Produces(MediaType.APPLICATION_JSON)
@Consumes(MediaType.APPLICATION_JSON)
public class NoteResource {

    @GET
    public List<Note> list() {
        return Note.listAll();
    }

    @GET
    @Path("/{id}")
    public Response getById(@PathParam("id") Long id) {
        Note note = Note.findById(id);
        if (note == null) {
            return Response.status(404).build();
        }
        return Response.ok(note).build();
    }

    @POST
    @Transactional
    public Response create(Note note) {
        note.persist();
        return Response.status(201).entity(note).build();
    }

    @PUT
    @Path("/{id}")
    @Transactional
    public Response update(@PathParam("id") Long id, Note updatedNote) {
        Note note = Note.findById(id);
        if (note == null) {
            return Response.status(404).build();
        }
        note.title = updatedNote.title;
        note.content = updatedNote.content;
        return Response.ok(note).build();
    }

    @DELETE
    @Path("/{id}")
    @Transactional
    public Response delete(@PathParam("id") Long id) {
        boolean deleted = Note.deleteById(id);
        return deleted ? Response.noContent().build()
                       : Response.status(404).build();
    }

    @GET
    @Path("/{id}/history")
    public Response getHistory(@PathParam("id") Long id) {
        Note note = Note.findById(id);
        if (note == null) {
            return Response.status(404).build();
        }
        // History-Logik hier
        return Response.ok(/* history list */).build();
    }
}

*Beispiel Angular Component (note-list.component.ts):*

[source,typescript]
----
import { Component, OnInit } from '@angular/core';
import { MessageService } from 'primeng/api';

interface Note {
  id?: number;
  title?: string;
  content?: string;
  createdAt?: string;
  updatedAt?: string;
}

interface NoteService {
  list(): any;
  update(id: number, note: Note): any;
  create(note: Note): any;
  delete(id: number): any;
}

@Component({
  selector: 'app-note-list',
  templateUrl: './note-list.component.html',
  styleUrls: ['./note-list.component.scss'],
  providers: [MessageService]
})
export class NoteListComponent implements OnInit {
  notes: Note[] = [];
  displayDialog = false;
  selectedNote?: Note;

  constructor(
    private noteService: NoteService,
    private messageService: MessageService
  ) {}

  ngOnInit(): void {
    this.loadNotes();
  }

  loadNotes(): void {
    this.noteService.list().subscribe({
      next: (data: Note[]) => this.notes = data,
      error: () => this.messageService.add({
        severity: 'error',
        summary: 'Fehler',
        detail: 'Notizen konnten nicht geladen werden'
      })
    });
  }

  showDialogToAdd(): void {
    this.selectedNote = {};
    this.displayDialog = true;
  }

  save(): void {
    if (this.selectedNote?.id) {
      this.noteService.update(this.selectedNote.id, this.selectedNote).subscribe({
        next: () => {
          this.messageService.add({
            severity: 'success',
            summary: 'Erfolg',
            detail: 'Notiz aktualisiert'
          });
          this.loadNotes();
          this.displayDialog = false;
        }
      });
    } else {
      this.noteService.create(this.selectedNote!).subscribe({
        next: () => {
          this.messageService.add({
            severity: 'success',
            summary: 'Erfolg',
            detail: 'Notiz erstellt'
          });
          this.loadNotes();
          this.displayDialog = false;
        }
      });
    }
  }

  delete(id: number): void {
    this.noteService.delete(id).subscribe({
      next: () => {
        this.messageService.add({
          severity: 'success',
          summary: 'Erfolg',
          detail: 'Notiz gelöscht'
        });
        this.loadNotes();
      }
    });
  }
}
